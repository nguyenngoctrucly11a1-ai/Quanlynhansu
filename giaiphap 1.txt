    function rankMatches(candidateRank,ranks){
      const a=String(candidateRank||"").toLowerCase();
      return ranks.some(r=>{const b=r.toLowerCase();return a===b||a.includes(b)||b.includes(a);});

    function typeMatches(experienceOrType, vesselType){
      const a=String(experienceOrType||"").toLowerCase(), b=String(vesselType||"").toLowerCase();
      if(!a||!b)return true;
      if(b.includes("chemical")) return a.includes("chemical");
      if(b.includes("tanker")) return a.includes("tanker");
      return a.includes(b);
    }

    function englishScore(){
      const ids=["engVocab","engReading","engListening","engSpeaking"];
      const values=ids.map(id=>Math.min(100,Math.max(0,Number(document.getElementById(id).value)||0)));
      const score=Math.round(values.reduce((a,b)=>a+b,0)/values.length);
      document.getElementById("englishScore").textContent=score+"/100";
      state.englishScore=score;
      const req=document.getElementById("english").value;
      const threshold=req.includes("Tốt")?80:req.includes("Khá")?65:50;
      const status=document.getElementById("englishStatus");
      status.className="badge "+(score>=threshold?"good":"warn");
      status.textContent=score>=threshold?"Đạt theo ngưỡng tham chiếu":"Chưa đạt – cần xem xét";
      return {score,threshold,pass:score>=threshold};
    }
